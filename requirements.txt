# ============================================================================
# 概率论伴学助手 · 原型依赖
# ----------------------------------------------------------------------------
# 安装：
#     pip install -r requirements.txt
#
# 说明：
#   - 前 8 项是**跑起来必需**的（Mock 模式下也能完整演示，不需要 API Key）。
#   - langchain-openai 只有连真实模型（DeepSeek / Qwen / GLM）时才用得上，
#     但体积不大，建议一并装上，免得演示当天才发现缺包。
#   - 版本用 >= 下界 + < 上界的宽松约束：既能拿到修 bug 的小版本，
#     又不会被未来的大版本破坏（本项目在 langgraph 1.x 上验证过）。
# ============================================================================

# ---- Web 服务 ----
fastapi>=0.115,<1.0
uvicorn[standard]>=0.30,<1.0
pydantic>=2.7,<3.0
pydantic-settings>=2.3,<3.0
python-dotenv>=1.0,<2.0

# ---- 认证（账号密码只存哈希）----
# 直接用 bcrypt，**不用 passlib**：我们只需要 hashpw / checkpw 两个函数，
# 而 passlib 1.7.4 读 bcrypt 4.x 的 __about__ 会抛 AttributeError（噪音大）。
bcrypt>=4.1,<5.0

# ---- 符号计算（分布性质推导，数学正确性的根基）----
sympy>=1.12,<2.0

# ---- 多 Agent 编排 ----
langgraph>=1.0,<2.0
langchain-core>=1.0,<2.0

# ---- 真实模型接入（OpenAI 兼容端点：DeepSeek / 通义 Qwen / 智谱 GLM）----
langchain-openai>=1.0,<2.0

# ============================================================================
# 不需要的东西（本项目刻意没用的，列出来避免有人画蛇添足）：
#   - 不需要 Node.js / npm：前端是免构建 ESM，KaTeX 已 vendor 到本地
#   - 不需要 Docker：存储用内置 sqlite3，向量库尚未接入（P2 阶段才需要 Qdrant）
#   - 不需要 matplotlib：分布图由前端 Canvas 自绘，后端只出数据
#   - 不需要 jieba 等分词库：中文检索用 bigram，标准库足够
# ============================================================================
